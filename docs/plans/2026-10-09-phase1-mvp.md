```python
"""JSONL 事件流读写：原子追加、半行截断、seq 推导（设计 §14.17 / §16.4 / §18）。"""

from __future__ import annotations

import json
import os
from pathlib import Path
from typing import Any


def read_events(path: Path) -> list[dict[str, Any]]:
    if not path.exists():
        return []
    text = path.read_text("utf-8")
    events: list[dict[str, Any]] = []
    for line in text.split("\n"):
        line = line.strip()
        if not line:
            continue
        try:
            events.append(json.loads(line))
        except json.JSONDecodeError:
            break  # 崩溃产生的半行：截断到最后一个完整事件（§16.4）
    return events


def next_seq(path: Path) -> int:
    events = read_events(path)
    return int(events[-1].get("seq", 0)) + 1 if events else 1


def append_event(path: Path, event: dict[str, Any]) -> dict[str, Any]:
    """按 seq 顺序原子追加（追加写 + flush + fsync），返回最终事件。"""
    path.parent.mkdir(parents=True, exist_ok=True)
    seq = next_seq(path)
    record = {"seq": seq, **event}
    line = json.dumps(record, ensure_ascii=False, separators=(",", ":")) + "\n"
    with open(path, "a", encoding="utf-8") as fh:
        fh.write(line)
        fh.flush()
        os.fsync(fh.fileno())
    return record
```

```python
"""DuckDB 连接封装（单写连接；原始表 src_*）。"""

from __future__ import annotations

from pathlib import Path
from typing import Any, Iterable, Sequence

import duckdb


def quote_ident(name: str) -> str:
    return '"' + name.replace('"', '""') + '"'


class Database:
    def __init__(self, path: Path) -> None:
        path.parent.mkdir(parents=True, exist_ok=True)
        self.path = path
        self._con = duckdb.connect(str(path))

    def execute(self, sql: str, params: Sequence[Any] | None = None):
        return self._con.execute(sql, params or [])

    def query(self, sql: str, params: Sequence[Any] | None = None):
        return self._con.execute(sql, params or [])

    def table_exists(self, table: str) -> bool:
        rows = self._con.execute(
            "SELECT 1 FROM information_schema.tables WHERE table_name = ?", [table]
        ).fetchall()
        return bool(rows)

    def create_table_from_rows(
        self, table: str, columns: Sequence[str], rows: Iterable[Sequence[Any]]
    ) -> int:
        cols = ", ".join(f"{quote_ident(c)} VARCHAR" for c in columns)
        self._con.execute(f"CREATE OR REPLACE TABLE {quote_ident(table)} ({cols})")
        placeholders = ", ".join("?" for _ in columns)
        batch = [tuple(r) for r in rows]
        if batch:
            self._con.executemany(
                f"INSERT INTO {quote_ident(table)} VALUES ({placeholders})", batch
            )
        return len(batch)

    def drop_table(self, table: str) -> None:
        if self.table_exists(table):
            self._con.execute(f"DROP TABLE {quote_ident(table)}")

    def close(self) -> None:
        self._con.close()
```

```python
"""Excel/CSV 读取（全字符串，保留原值；设计 §14.3）。"""

from __future__ import annotations

from pathlib import Path

import pandas as pd


def read_table(path: Path) -> tuple[list[str], list[list[str]]]:
    suffix = path.suffix.lower()
    if suffix in (".xlsx", ".xlsm"):
        frame = pd.read_excel(path, dtype=str, engine="openpyxl").fillna("")
    elif suffix in (".csv", ".txt"):
        try:
            frame = pd.read_csv(path, dtype=str, keep_default_na=False, encoding="utf-8-sig")
        except UnicodeDecodeError:
            frame = pd.read_csv(path, dtype=str, keep_default_na=False, encoding="gbk")
    else:
        raise ValueError(f"unsupported file type: {suffix}")
    columns = [str(c) for c in frame.columns]
    rows = [[str(v) for v in row] for row in frame.itertuples(index=False, name=None)]
    return columns, rows
```

```python
"""列类型推断（采样；仅用于 schema.json 标注，落库仍为 VARCHAR）。"""

from __future__ import annotations

import re

_DECIMAL = re.compile(r"^-?\d+(\.\d+)?$")
_DATE = re.compile(r"^\d{4}[-/]\d{1,2}[-/]\d{1,2}")


def infer_type(sample: str) -> str:
    value = (sample or "").strip()
    if not value:
        return "string"
    if _DATE.match(value):
        return "date"
    if _DECIMAL.match(value):
        return "decimal"
    return "string"


def infer_column(column: str, values: list[str]) -> dict[str, str]:
    samples = [v for v in values if v and v.strip()][:50]
    inferred = "string"
    if samples:
        types = {infer_type(v) for v in samples}
        inferred = types.pop() if len(types) == 1 else "string"
    first = samples[0] if samples else ""
    return {"name": column, "type": inferred, "sample": first}
```

```python
"""索引文件读写：schema.json（表结构摘要）、files.json（文件指纹）。"""

from __future__ import annotations

import json
from pathlib import Path
from typing import Any


def _read(path: Path, default):
    if not path.exists():
        return default
    try:
        return json.loads(path.read_text("utf-8"))
    except (OSError, json.JSONDecodeError):
        return default


def _write(path: Path, data) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(data, ensure_ascii=False, indent=2), "utf-8")


def load_schema(path: Path) -> list[dict[str, Any]]:
    return _read(path, [])


def save_schema(path: Path, entries: list[dict[str, Any]]) -> None:
    _write(path, entries)


def upsert_schema_entry(path: Path, entry: dict[str, Any]) -> None:
    entries = [e for e in load_schema(path) if e.get("table") != entry.get("table")]
    entries.append(entry)
    save_schema(path, entries)


def load_files(path: Path) -> list[dict[str, Any]]:
    return _read(path, [])


def save_files(path: Path, entries: list[dict[str, Any]]) -> None:
    _write(path, entries)
```

`import_source(root, path, *, progress=None)` 逻辑：
1. 计算文件 sha256；读 `files.json`，若已存在同 sha256 且 `status=ok` → `raise ApiError("E_IMPORT_DUP", 409, "error.importDup", {"path": str(path)})`。

6. 更新 `project.json`：追加 `dataSources` 项、`counters.dataSource = N`、`updatedAt`。

```python
"""SQL 只读沙箱（设计 §17.3）。"""

from __future__ import annotations

import re
from ..errors import ApiError

_ALLOWED = ("select", "with", "explain", "describe")
_FORBIDDEN = re.compile(
    r"\b(attach|detach|copy|install|load|export|pragma|create|drop|alter|insert|update|delete|truncate|call|set)\b",
    re.IGNORECASE,
)


def validate_sql(sql: str) -> str:
    text = sql.strip().rstrip(";").strip()
    if not text:
        raise ApiError("E_SQL_REJECTED", 400, "error.sqlRejected", {"reason": "empty"})
    if ";" in text:
        raise ApiError("E_SQL_REJECTED", 400, "error.sqlRejected", {"reason": "multiple_statements"})
    first = re.split(r"\s+", text, maxsplit=1)[0].lower()
    if first not in _ALLOWED:
        raise ApiError("E_SQL_REJECTED", 400, "error.sqlRejected", {"reason": "not_readonly"})
    if _FORBIDDEN.search(text):
        raise ApiError("E_SQL_REJECTED", 400, "error.sqlRejected", {"reason": "forbidden_keyword"})
    return text
```

`tools/sql_query.py`：`validate_sql(sql)` → 注入 `LIMIT`（若语句无 limit，包一层 `SELECT * FROM (sql) LIMIT n+1`）→ 执行，超时用 `con.interrupt` 或线程超时；`rowCount > limit` → `truncated=True` 并 `dump_artifact` 返回 `artifactRef`。默认 `limit=200`（工具级），上限 `10000`。执行异常转 `ApiError("E_SQL_REJECTED"|"E_SQL_TIMEOUT")`。

## 风险与注意事项

- **Excel 大文件**：pandas 全量读入内存；若近 100 万行需分块（chunk）导入。Phase 1 先设单文件上限校验（≤200MB），超限提示。
- **DuckDB 单写连接**：导入与查询串行化；导入进行中拒绝并发查询。
- **模型不确定输出**：`GenericFakeChatModel` 仅用于单测；集成测试避免依赖真实模型。
- **append-only 语义**：`findings.jsonl` 不物理删除（§14.8）。
- **`__row_id` 稳定性**：重导复用策略在 Phase 1 先「整表替换、row_id 重新生成」，跨重导定位精确性作为 Phase 2 增强。
